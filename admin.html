🗄️ Setup SQL Supabase (klik untuk lihat) 
-- Jalankan sekali di Supabase SQL Editor:

CREATE TABLE IF NOT EXISTS site_stats (
  page        text PRIMARY KEY,
  views       bigint DEFAULT 0,
  last_seen   timestamptz DEFAULT now()
);
CREATE TABLE IF NOT EXISTS site_daily (
  day   date NOT NULL,
  page  text NOT NULL,
  views bigint DEFAULT 0,
  PRIMARY KEY (day, page)
);

-- Seed 10 halaman
INSERT INTO site_stats (page) VALUES
  ('home'),

  ('novel-list'),('novel-detail'),('novel-read'),
  ('anime-list'),('anime-detail'),('anime-watch')
ON CONFLICT DO NOTHING;

ALTER TABLE site_stats ENABLE ROW LEVEL SECURITY;

CREATE POLICY "anon_all_stats" ON site_stats FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "anon_all_daily" ON site_daily FOR ALL USING (true) WITH CHECK (true);
